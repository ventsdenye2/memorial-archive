var root=MemorialArchive.Framework.Core.GameRoot.Instance;
if(!UnityEditor.EditorApplication.isPlaying || root==null) return "Not ready";
var view=UnityEngine.Object.FindObjectOfType<MemorialArchive.Framework.UI.InteractionPromptView>();
var flags=System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance;
var type=view.GetType();
var prompt=(UnityEngine.GameObject)type.GetField("promptRoot",flags).GetValue(view);
var image=(UnityEngine.UI.Image)type.GetField("promptImage",flags).GetValue(view);
var button=(UnityEngine.UI.Button)type.GetField("promptButton",flags).GetValue(view);
var lighting=root.GetSystem<MemorialArchive.Gameplay.Lighting.Logic.LightingSystem>();
var flow=root.GetSystem<MemorialArchive.Gameplay.Guide.Logic.GuideFlowSystem>();
root.Context.UI.CloseAll();
var stairs=UnityEngine.Object.FindObjectsOfType<MemorialArchive.Gameplay.Interaction.View.InteractionPointView>().First(p=>root.Context.Configs.GetInteraction(p.InteractionId)!=null && root.Context.Configs.GetInteraction(p.InteractionId).HasStairDestinations);
root.Context.Events.Publish(new MemorialArchive.Framework.Event.InteractionFocusChangedEvent(stairs.InteractionId,MemorialArchive.Gameplay.Interaction.Data.InteractionType.SceneExit,true));
type.GetMethod("Update",flags).Invoke(view,null);
bool stairsVisible=prompt.activeSelf;
bool stairsClickable=button.interactable;
int travels=0;
System.Action<MemorialArchive.Framework.Event.StairTravelRequestedEvent> onTravel=e=>travels++;
root.Context.Events.Subscribe(onTravel);
button.onClick.Invoke();
root.Context.Events.Unsubscribe(onTravel);
root.Context.UI.CloseAll();
typeof(MemorialArchive.Gameplay.Guide.Logic.GuideFlowSystem).GetField("scene",flags).SetValue(flow,"FrontHall");
typeof(MemorialArchive.Gameplay.Guide.Logic.GuideFlowSystem).GetField("playerPosition",flags).SetValue(flow,new UnityEngine.Vector2(flow.Config.frontHallLeft+flow.Config.equipmentGatePixels/100f-0.6f,-4.6f));
type.GetMethod("Update",flags).Invoke(view,null);
var pickup=(UnityEngine.Sprite)type.GetField("pickupSprite",flags).GetValue(view);
bool chairVisible=prompt.activeSelf;
bool chairPickup=image.sprite==pickup && pickup!=null;
bool chairClickable=button.interactable;
bool chairHandlesInteraction=flow.TryInteract();
return new{lanternEquipped=lighting.IsLanternEquipped,stairsVisible,stairsClickable,travels,chairVisible,chairPickup,chairClickable,chairHandlesInteraction};
