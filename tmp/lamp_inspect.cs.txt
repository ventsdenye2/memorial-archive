var camera = UnityEngine.Camera.main;
var result = new System.Collections.Generic.List<object>();
foreach(var lamp in UnityEngine.Object.FindObjectsOfType<MemorialArchive.Gameplay.Lighting.View.LightSourceView>(true)) {
 if (!lamp.name.Contains("special")) continue;
 var sr=lamp.GetComponent<UnityEngine.SpriteRenderer>();
 result.Add(new {name=lamp.name, active=lamp.gameObject.activeInHierarchy, scene=lamp.gameObject.scene.name, special=lamp.IsSpecial, visible=lamp.IsVisualVisible, position=lamp.transform.position.ToString(), center=sr.bounds.center.ToString(), screen=camera==null?"none":camera.WorldToScreenPoint(sr.bounds.center).ToString()});
}
var root=MemorialArchive.Framework.Core.GameRoot.Instance;
var guide=root==null?null:root.GetSystem<MemorialArchive.Gameplay.Guide.Logic.GuideSystem>();
var flow=root==null?null:root.GetSystem<MemorialArchive.Gameplay.Guide.Logic.GuideFlowSystem>();
return new {project=UnityEngine.Application.dataPath, playing=UnityEditor.EditorApplication.isPlaying, screen=new[]{UnityEngine.Screen.width,UnityEngine.Screen.height}, camera=camera==null?"none":camera.transform.position.ToString(), lamps=result, hasPosition=flow==null?false:flow.HasPosition, progress=guide==null?"none":UnityEngine.JsonUtility.ToJson(guide.CaptureSaveData())};
