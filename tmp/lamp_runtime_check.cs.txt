var lamp=UnityEngine.GameObject.Find("light_fronthall_special");

return new{position=camera.WorldToScreenPoint(lamp.transform.position).ToString(),current=guide.Current==null?"none":guide.Current.id,blocked=root.Context.UI.IsGameplayInputBlocked};
